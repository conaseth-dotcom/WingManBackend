/**
 * safeMount(mountPath, routerModule, label)
 * Accepts: express.Router, { router }, or a plain middleware function.
 */
function safeMount(mountPath, mod, label) {
  if (!mod) return;
  try {
    const router = mod.router || mod.default || mod;
    if (typeof router === 'function') {
      app.use(mountPath, router);
      logger.info(`Route mounted: ${mountPath}  [${label}]`);
    }
  } catch (e) {
    logger.warn(`Route mount failed: ${mountPath} [${label}]`, { err: e.message });
  }
}

// ── Core server routes (barrel or glob) ──────────────────────────────────────
try {
  const coreRoutes = require(r('core/server/routes'));
  safeMount('/api', coreRoutes, 'core-routes');
} catch {
  const glob  = require('glob');
  const files = glob.sync(r('core/server/routes/**/*.route.cjs').replace(/\\/g, '/'));
  for (const f of files) {
    try {
      const slug = path.basename(f, '.route.cjs');
      safeMount(`/api/${slug}`, require(f), `core-route:${slug}`);
    } catch (e) { logger.warn(`Core route load failed: ${f}`, { err: e.message }); }
  }
}

const SHUTDOWN_TIMEOUT_MS = 10_000;

async function gracefulShutdown(signal) {
  logger.info(`Shutdown signal: ${signal}`);
  eventBus.emit('server:shutdown', { signal });
  app.locals.ready = false;

  const deadline = setTimeout(() => {
    logger.error('Shutdown timed out — forcing exit');
    process.exit(1);
  }, SHUTDOWN_TIMEOUT_MS).unref();

  // Reverse startup order
  const teardowns = [
    // Phase 5 — Feature Systems
    ['Trays',          trayLoader,       'shutdown'],
    ['Access',         accessManager,    'shutdown'],
    ['Partition',      partitionStorage, 'shutdown'],
    // Phase 4 — I/O
    ['Whisper',        whisperEngine,    'shutdown'],
    ['VoiceTransport', voiceTransport,   'shutdown'],
    ['Mic',            micEngine,        'shutdown'],
    // Phase 3 — BlackBox
    ['BB Gateway',     bbGateway,        'shutdown'],
    ['BB UI',          bbUI,             'shutdown'],
    ['BB Relation',    bbRelationship,   'shutdown'],
    ['BB Thinking',    bbThinking,       'shutdown'],
    ['BB Autonomy',    bbAutonomy,       'shutdown'],
    ['BB Runtime',     bbRuntime,        'shutdown'],
    // Phase 2 — AI Stack
    ['ContextBuilder', contextBuilder,   'shutdown'],
    ['Memory',         memory,           'shutdown'],
    ['Identity',       identity,         'shutdown'],
    ['Personality',    personality,      'shutdown'],
    ['AI Pipeline',    aiPipeline,       'shutdown'],
    ['AI Runtime',     aiRuntime,        'shutdown'],
  ];

  for (const [label, mod, method] of teardowns) {
    if (mod && typeof mod[method] === 'function') {
      try {
        await Promise.resolve(mod[method]());
        logger.info(`Shutdown: ${label} ✓`);
      } catch (e) {
        logger.warn(`Shutdown: ${label} error`, { err: e.message });
      }
    }
  }

  if (xttsProcess && !xttsProcess.killed) {
    xttsProcess.kill('SIGTERM');
    logger.info('Shutdown: XTTS process terminated');
  }

const SHUTDOWN_TIMEOUT_MS = 10_000;

async function gracefulShutdown(signal) {
  logger.info(`Shutdown signal: ${signal}`);
  eventBus.emit('server:shutdown', { signal });
  app.locals.ready = false;

  const deadline = setTimeout(() => {
    logger.error('Shutdown timed out — forcing exit');
    process.exit(1);
  }, SHUTDOWN_TIMEOUT_MS).unref();

  // Reverse startup order
  const teardowns = [
    // Phase 5 — Feature Systems
    ['Trays',          trayLoader,       'shutdown'],
    ['Access',         accessManager,    'shutdown'],
    ['Partition',      partitionStorage, 'shutdown'],
    // Phase 4 — I/O
    ['Whisper',        whisperEngine,    'shutdown'],
    ['VoiceTransport', voiceTransport,   'shutdown'],
    ['Mic',            micEngine,        'shutdown'],
    // Phase 3 — BlackBox
    ['BB Gateway',     bbGateway,        'shutdown'],
    ['BB UI',          bbUI,             'shutdown'],
    ['BB Relation',    bbRelationship,   'shutdown'],
    ['BB Thinking',    bbThinking,       'shutdown'],
    ['BB Autonomy',    bbAutonomy,       'shutdown'],
    ['BB Runtime',     bbRuntime,        'shutdown'],
    // Phase 2 — AI Stack
    ['ContextBuilder', contextBuilder,   'shutdown'],
    ['Memory',         memory,           'shutdown'],
    ['Identity',       identity,         'shutdown'],
    ['Personality',    personality,      'shutdown'],
    ['AI Pipeline',    aiPipeline,       'shutdown'],
    ['AI Runtime',     aiRuntime,        'shutdown'],
  ];

  for (const [label, mod, method] of teardowns) {
    if (mod && typeof mod[method] === 'function') {
      try {
        await Promise.resolve(mod[method]());
        logger.info(`Shutdown: ${label} ✓`);
      } catch (e) {
        logger.warn(`Shutdown: ${label} error`, { err: e.message });
      }
    }
  }

  if (xttsProcess && !xttsProcess.killed) {
    xttsProcess.kill('SIGTERM');
    logger.info('Shutdown: XTTS process terminated');
  }

  await new Promise((resolve) => server.close(resolve));
  logger.info('HTTP server closed');

  clearTimeout(deadline);
  logger.info('WingMan Orchestrator exited cleanly');
  process.exit(0);
}

const SHUTDOWN_TIMEOUT_MS = 10_000;

async function gracefulShutdown(signal) {
  logger.info(`Shutdown signal: ${signal}`);
  eventBus.emit('server:shutdown', { signal });
  app.locals.ready = false;

  const deadline = setTimeout(() => {
    logger.error('Shutdown timed out — forcing exit');
    process.exit(1);
  }, SHUTDOWN_TIMEOUT_MS).unref();

  // Reverse startup order
  const teardowns = [
    // Phase 5 — Feature Systems
    ['Trays',          trayLoader,       'shutdown'],
    ['Access',         accessManager,    'shutdown'],
    ['Partition',      partitionStorage, 'shutdown'],
    // Phase 4 — I/O
    ['Whisper',        whisperEngine,    'shutdown'],
    ['VoiceTransport', voiceTransport,   'shutdown'],
    ['Mic',            micEngine,        'shutdown'],
    // Phase 3 — BlackBox
    ['BB Gateway',     bbGateway,        'shutdown'],
    ['BB UI',          bbUI,             'shutdown'],
    ['BB Relation',    bbRelationship,   'shutdown'],
    ['BB Thinking',    bbThinking,       'shutdown'],
    ['BB Autonomy',    bbAutonomy,       'shutdown'],
    ['BB Runtime',     bbRuntime,        'shutdown'],
    // Phase 2 — AI Stack
    ['ContextBuilder', contextBuilder,   'shutdown'],
    ['Memory',         memory,           'shutdown'],
    ['Identity',       identity,         'shutdown'],
    ['Personality',    personality,      'shutdown'],
    ['AI Pipeline',    aiPipeline,       'shutdown'],
    ['AI Runtime',     aiRuntime,        'shutdown'],
  ];

  for (const [label, mod, method] of teardowns) {
    if (mod && typeof mod[method] === 'function') {
      try {
        await Promise.resolve(mod[method]());
        logger.info(`Shutdown: ${label} ✓`);
      } catch (e) {
        logger.warn(`Shutdown: ${label} error`, { err: e.message });
      }
    }
  }

  if (xttsProcess && !xttsProcess.killed) {
    xttsProcess.kill('SIGTERM');
    logger.info('Shutdown: XTTS process terminated');
  }

  await new Promise((resolve) => server.close(resolve));
  logger.info('HTTP server closed');

  clearTimeout(deadline);
  logger.info('WingMan Orchestrator exited cleanly');
  process.exit(0);
}

  logger.error('Uncaught exception', { err: err.message, stack: err.stack });
  gracefulShutdown('uncaughtException');
});

const SHUTDOWN_TIMEOUT_MS = 10_000;

async function gracefulShutdown(signal) {
  logger.info(`Shutdown signal: ${signal}`);
  eventBus.emit('server:shutdown', { signal });
  app.locals.ready = false;

  const deadline = setTimeout(() => {
    logger.error('Shutdown timed out — forcing exit');
    process.exit(1);
  }, SHUTDOWN_TIMEOUT_MS).unref();

  // Reverse startup order
  const teardowns = [
    // Phase 5 — Feature Systems
    ['Trays',          trayLoader,       'shutdown'],
    ['Access',         accessManager,    'shutdown'],
    ['Partition',      partitionStorage, 'shutdown'],
    // Phase 4 — I/O
    ['Whisper',        whisperEngine,    'shutdown'],
    ['VoiceTransport', voiceTransport,   'shutdown'],
    ['Mic',            micEngine,        'shutdown'],
    // Phase 3 — BlackBox
    ['BB Gateway',     bbGateway,        'shutdown'],
    ['BB UI',          bbUI,             'shutdown'],
    ['BB Relation',    bbRelationship,   'shutdown'],
    ['BB Thinking',    bbThinking,       'shutdown'],
    ['BB Autonomy',    bbAutonomy,       'shutdown'],
    ['BB Runtime',     bbRuntime,        'shutdown'],
    // Phase 2 — AI Stack
    ['ContextBuilder', contextBuilder,   'shutdown'],
    ['Memory',         memory,           'shutdown'],
    ['Identity',       identity,         'shutdown'],
    ['Personality',    personality,      'shutdown'],
    ['AI Pipeline',    aiPipeline,       'shutdown'],
    ['AI Runtime',     aiRuntime,        'shutdown'],
  ];

  for (const [label, mod, method] of teardowns) {
    if (mod && typeof mod[method] === 'function') {
      try {
        await Promise.resolve(mod[method]());
        logger.info(`Shutdown: ${label} ✓`);
      } catch (e) {
        logger.warn(`Shutdown: ${label} error`, { err: e.message });
      }
    }
  }

  if (xttsProcess && !xttsProcess.killed) {
    xttsProcess.kill('SIGTERM');
    logger.info('Shutdown: XTTS process terminated');
  }

  await new Promise((resolve) => server.close(resolve));
  logger.info('HTTP server closed');

  clearTimeout(deadline);
  logger.info('WingMan Orchestrator exited cleanly');
  process.exit(0);
}

  logger.error('Uncaught exception', { err: err.message, stack: err.stack });

/**
 * safeMount(mountPath, routerModule, label)
 * Accepts: express.Router, { router }, or a plain middleware function.
 */
function safeMount(mountPath, mod, label) {
  if (!mod) return;
  try {
    const router = mod.router || mod.default || mod;
    if (typeof router === 'function') {
      app.use(mountPath, router);
      logger.info(`Route mounted: ${mountPath}  [${label}]`);
    }
  } catch (e) {
    logger.warn(`Route mount failed: ${mountPath} [${label}]`, { err: e.message });
  }
}

const SHUTDOWN_TIMEOUT_MS = 10_000;

async function gracefulShutdown(signal) {
  logger.info(`Shutdown signal: ${signal}`);
  eventBus.emit('server:shutdown', { signal });
  app.locals.ready = false;

  const deadline = setTimeout(() => {
    logger.error('Shutdown timed out — forcing exit');
    process.exit(1);
  }, SHUTDOWN_TIMEOUT_MS).unref();

  // Reverse startup order
  const teardowns = [
    // Phase 5 — Feature Systems
    ['Trays',          trayLoader,       'shutdown'],
    ['Access',         accessManager,    'shutdown'],
    ['Partition',      partitionStorage, 'shutdown'],
    // Phase 4 — I/O
    ['Whisper',        whisperEngine,    'shutdown'],
    ['VoiceTransport', voiceTransport,   'shutdown'],
    ['Mic',            micEngine,        'shutdown'],
    // Phase 3 — BlackBox
    ['BB Gateway',     bbGateway,        'shutdown'],
    ['BB UI',          bbUI,             'shutdown'],
    ['BB Relation',    bbRelationship,   'shutdown'],
    ['BB Thinking',    bbThinking,       'shutdown'],
    ['BB Autonomy',    bbAutonomy,       'shutdown'],
    ['BB Runtime',     bbRuntime,        'shutdown'],
    // Phase 2 — AI Stack
    ['ContextBuilder', contextBuilder,   'shutdown'],
    ['Memory',         memory,           'shutdown'],
    ['Identity',       identity,         'shutdown'],
    ['Personality',    personality,      'shutdown'],
    ['AI Pipeline',    aiPipeline,       'shutdown'],
    ['AI Runtime',     aiRuntime,        'shutdown'],
  ];

  for (const [label, mod, method] of teardowns) {
    if (mod && typeof mod[method] === 'function') {
      try {
        await Promise.resolve(mod[method]());
        logger.info(`Shutdown: ${label} ✓`);
      } catch (e) {
        logger.warn(`Shutdown: ${label} error`, { err: e.message });
      }
    }
  }

  if (xttsProcess && !xttsProcess.killed) {
    xttsProcess.kill('SIGTERM');
    logger.info('Shutdown: XTTS process terminated');
  }

  await new Promise((resolve) => server.close(resolve));
  logger.info('HTTP server closed');

  clearTimeout(deadline);
  logger.info('WingMan Orchestrator exited cleanly');
  process.exit(0);
}
